=== 1. PAGES (new files under shared app routing) ===
The panels are client components that take canManage (fail-closed default false). Server must pass it.

--- app/onlyfans/launch/automation/page.tsx (new)
import BAutomationPanel from "@/components/onlyfans-launch/automation/BAutomationPanel";
import { getDashboardAccess } from "@/lib/dashboard-access";
import { agencyCanManage } from "@/lib/agency-management-policy";
import { isAuthed } from "@/lib/auth";
export const dynamic = "force-dynamic";
export default async function Page() {
  const agency = await getDashboardAccess();
  const canManage = agency ? agencyCanManage(agency) : await isAuthed();
  return <BAutomationPanel canManage={canManage} />;
}
